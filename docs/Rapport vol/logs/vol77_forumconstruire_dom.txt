	function sh_defil_avis()
		{
		$('#hp_news li:first').hide('slow', function()
			{
			$('#hp_news').find("li:last").after($('#hp_news').find("li:first"));
			$('#hp_news li:last').show();
			});

	function sh_defil_toogle(element)
		{
		if(hp_news_toogle_state == 0)
			{
			hp_news_toogle_state = 1;
			hp_new_defil_2 = false;
			$('#hp_news').addClass('hp_news_fixed');
			$('#hp_news').removeClass('hp_news_defil');
			element.html('Reduire la fenêtre');
			$('#hp_news_loader').css('animation', 'none');
			}
		else
			{
			hp_new_defil_2 = true;
			$('#hp_news').addClass('hp_news_defil');
			$('#hp_news').removeClass('hp_news_fixed');
			hp_news_toogle_state = 0;
			element.html('Voir toutes les actus');
			haut = $('#hp_news').offset();
			to_top = haut.top-150;
			window.scrollTo(0, to_top);
			}
		}
